<h1 id="title">Response</h1>
<form>
  <label id="prompt" for="notes">What would you change?</label>
  <textarea id="notes" required></textarea>
  <button type="submit">Send</button>
  <button id="cancel" type="button">Cancel</button>
</form>
<p id="delivery"></p>
<p id="error" role="alert"></p>
<script type="module">
  const form = document.querySelector('form');
  const notes = document.querySelector('#notes');
  const error = document.querySelector('#error');
  let busy = false;
  form.inert = true;
  try {
    const {input, delivery} = await fungi.ready();
    document.title = document.querySelector('#title').textContent = input?.title ?? 'Response';
    document.querySelector('#prompt').textContent = input?.prompt ?? 'What would you change?';
    if (delivery.mode === 'manual') {
      document.querySelector('#delivery').textContent = 'After sending, tell your agent “done”.';
    }
    async function finish(cancelled) {
      if (busy) return;
      busy = true; form.inert = true; error.textContent = '';
      try {
        if (cancelled) await fungi.cancel();
        else await fungi.submit({notes: notes.value});
      } catch (e) {
        error.textContent = e.message;
        busy = false; form.inert = false;
      }
    }
    form.onsubmit = event => { event.preventDefault(); finish(false); };
    document.querySelector('#cancel').onclick = () => finish(true);
    form.inert = false;
  } catch (e) { error.textContent = e.message; }
</script>
